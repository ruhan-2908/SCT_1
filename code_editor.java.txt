import java.io.*;
import java.util.*;

public class Solution {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String input = sc.next();
        Stack<Character> stack = new Stack<>();
        for(char ch : input.toCharArray())
        {
            if(ch == ')' || ch == ']' || ch == '}')
            {
                if(stack.isEmpty()) 
                {
                    System.out.println(false);
                    return ;
                }
                else if((ch == ')' && stack.peek() != '(') || 
                (ch == ']' && stack.peek() != '[') || 
                (ch == '}' && stack.peek() != '{'))
                {
                    System.out.println(false);
                    return ;
                }else
                {
                    stack.pop();
                }
                
            }else
            {
                stack.push(ch);
            }
        }
        if(stack.isEmpty())
        {
            System.out.println(true);
            return ;
        }else
        {
            System.out.println(false);
        }
    }
}





































public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);

    int size = sc.nextInt();

    if (size <= 0) {
        System.out.print(0);
        return;
    }

    sc.nextLine(); 
    
    String[] home = sc.nextLine().trim().split(" ");

    if (home.length == 0 || home[0].equals("null")) {
        System.out.print(0);
        return;
    }

    int idx = 0;
    Node root = new Node(Integer.parseInt(home[idx++]));

    Queue<Node> q = new LinkedList<>();
    q.offer(root);

    while (idx < home.length && !q.isEmpty()) {
        Node curr = q.poll();

        if (idx < home.length && !home[idx].equals("null")) {
            curr.left = new Node(Integer.parseInt(home[idx]));
            q.offer(curr.left);
        }
        idx++;

        if (idx < home.length && !home[idx].equals("null")) {
            curr.right = new Node(Integer.parseInt(home[idx]));
            q.offer(curr.right);
        }
        idx++;
    }
}
