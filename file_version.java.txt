import java.io.*;
import java.util.*;

public class Solution {
    
    Queue<Integer> q1;
    Queue<Integer> q2;
    
    Solution()
    {
        q1 = new LinkedList<>();
        q2 = new LinkedList<>();
    }
    
    public void push(int data)
    {
        q1.offer(data);
    }
    
    public int pop()
    {
        while(q1.size() > 1)
        {
            q2.offer(q1.peek());
            q1.poll();
        }
        int val = q1.poll();
        swap();
        return val;
    }
    
    public int peek()
    {
        while(q1.size() > 1)
        {
            q2.offer(q1.peek());
            q1.poll();
        }
        int val = q1.peek();
        q2.offer(val);
        q1.poll();
        swap();
        return val;
    }
    
    public void swap()
    {
        Queue<Integer> temp = q1;
        q1 = q2;
        q2 = temp;
    }
    
    public boolean isEmpty()
    {
        return q1.isEmpty();
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        Solution st = new Solution();
        int size = sc.nextInt();
        for(int i = 0 ; i < size ; i++)
        {
            String op = sc.next();
            if(op.equals("push"))
            {
                int num = sc.nextInt();
                st.push(num);
                System.out.println("Pushed: " + num);
            }else if(op.equals("pop"))
            {
                int num = st.pop();
                System.out.println("Popped: " + num);
            }else if(op.equals("top"))
            {
                int num = st.peek();
                System.out.println("Top version: " + num);
            }else
            {
                System.out.println("Is stack empty: " + st.isEmpty());
            }
        }
    }
}
