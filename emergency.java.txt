import java.io.*;
import java.util.*;

public class Solution {

    public static void main(String[] args) {
        PriorityQueue<String[]> maxHeap = new PriorityQueue<>((a,b) -> Integer.compare(Integer.parseInt(b[1]),Integer.parseInt(a[1])));
        Scanner sc = new Scanner(System.in);
        int size = sc.nextInt();
        for(int i =0 ; i < size ; i++)
        {
            String op = sc.next();
            if(op.equals("ENQUEUE"))
            {
                String name = sc.next();
                String prio = sc.next();
                maxHeap.offer(new String[]{name,prio});
            }else if(op.equals("DEQUEUE"))
            {
                String[] curr = maxHeap.poll();
                System.out.println("Treated Patient: " + curr[0]);
            }else 
            {
                if(maxHeap.isEmpty())
                {
                    System.out.println("Queue is empty.");
                }else
                {
                    List<String[]> temp = new ArrayList<>();
                    System.out.print("Current Queue: ");
                    while(!maxHeap.isEmpty())
                    {
                        String[] curr = maxHeap.poll();
                        temp.add(curr);
                        System.out.print(curr[0] + '(' + curr[1] + ')' + " ");
                    }
                    System.out.println();
                    for(String[] t : temp) maxHeap.offer(t);
                    
                }
            }
        }
    }
}
